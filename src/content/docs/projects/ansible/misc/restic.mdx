---
title: restic
description: Install and configure restic/resticprofile for automated repository backups.
tableOfContents: false
ansibleRole:
  collection: giftpilz0.misc
  name: restic
  variables:
  - name: resticprofile_version
    type: str
    description: resticprofile release version.
    default: 0.33.1
  - name: rest_server_version
    type: str
    description: rest-server release version.
    default: 0.14.0
  - name: restic_mode
    type: str
    description: role operating mode.
    default: client
    choices:
    - client
    - server
    - both
  - name: restic_user
    type: str
    description: system user for restic operations.
    default: backup
  - name: restic_user_home
    type: str
    description: home directory of the restic user.
    default: /home/{{ restic_user }}
  - name: restic_config_dir
    type: str
    description: directory for resticprofile configuration.
    default: /etc/resticprofile
  - name: restic_cache_dir
    type: str
    description: directory for restic cache data.
    default: /var/cache/restic
  - name: restic_log_dir
    type: str
    description: directory for restic log files.
    default: '{{ restic_config_dir }}/logs'
  - name: restic_password_dir
    type: str
    description: directory for repository password files.
    default: '{{ restic_config_dir }}/passwords'
  - name: restic_env_dir
    type: str
    description: directory for the environment files of restic-repo and resticprofile.
    default: '{{ restic_config_dir }}/env'
  - name: restic_server_repo_base
    type: str
    description: base directory for server-side repositories.
    default: /var/lib/restic
  - name: restic_server_host
    type: str
    description: hostname or IP of the rest-server.
    default: '""'
  - name: restic_server_port
    type: int
    description: port of the rest-server.
    default: '8000'
  - name: restic_server_password
    type: str
    description: rest-server password of the client, the username is the inventory hostname.
    default: '""'
  - name: restic_stop_units_user
    type: str
    description: owner of the user units in stop_units of a repository.
    default: '""'
  - name: restic_repositories
    type: list[dict]
    description: list of restic repositories to manage.
    default: '[]'
    options:
    - name: name
      type: str
      description: repository name.
    - name: password
      type: str
      description: repository password.
    - name: backup_paths
      type: list[str]
      description: paths to back up.
    - name: exclude_patterns
      type: list[str]
      description: patterns to exclude from backup.
    - name: stop_units
      type: list[str]
      description: user units stopped in order during the backup and started in reverse.
    - name: schedule
      type: str
      description: systemd calendar schedule (e.g. *-*-* 02:00:00).
    - name: retention
      type: dict
      options:
      - name: keep_daily
        type: int
        description: number of daily snapshots to keep.
      - name: keep_weekly
        type: int
        description: number of weekly snapshots to keep.
      - name: keep_monthly
        type: int
        description: number of monthly snapshots to keep.
      - name: keep_yearly
        type: int
        description: number of yearly snapshots to keep.
  - name: restic_repository_defaults
    type: dict
    description: values for keys missing in restic_repositories entries (merged recursively).
    default: "exclude_patterns: []\nstop_units: []\nretention:\n  keep_daily: 14\n  keep_weekly: 8\n  keep_monthly: 12\n  keep_yearly: 3"
  - name: restic_restore_test_enabled
    type: bool
    description: run the weekly restore test.
    default: 'true'
  - name: restic_restore_test_schedule
    type: str
    description: systemd calendar schedule of the restore test.
    default: Sat 04:00
  - name: restic_restore_test_max_age_hours
    type: int
    description: the restore test fails when the latest snapshot is older.
    default: '36'
  - name: restic_restore_test_files
    type: int
    description: number of random files restored per repository.
    default: '20'
  - name: restic_restore_test_max_file_size
    type: int
    description: largest file in bytes picked for the restore test.
    default: '52428800'
---
