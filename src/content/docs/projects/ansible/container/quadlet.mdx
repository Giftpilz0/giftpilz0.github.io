    type: bool
    description: Configure an HTTP proxy for image pulls and all containers on the host.
    default: 'false'
  - name: podman_proxy_http
    type: str
    description: HTTP proxy URL.
    default: '""'
  - name: podman_proxy_https
    type: str
    description: HTTPS proxy URL.
    default: '""'
  - name: podman_proxy_no_proxy
    type: str
    description: Comma-separated list of proxy exclusions.
    default: '""'
  - name: quadlet_role
    type: str
    required: true
    description: Service role whose templates/ and files/ trees are deployed.
    default: '""'

    default: 'false'

    default: 'false'
  - name: quadlet_extra_files
    type: list[dict]
    description: Extra file entries with src (controller path) or source (host path), dest, optional mount, mode, template and relabel (SELinux :Z).
    default: '[]'
  - name: quadlet_extra_dirs
    type: list[dict]
    description: Extra directory entries with src (controller path) or source (host path), dest, optional mount, mode and relabel (SELinux :Z).
    default: '[]'
  - name: quadlet_extra_file_defaults
    type: dict
    description: Values for keys missing in quadlet_extra_files entries.
    default: 'mode: ''0640''

      template: false

      mount: ''''

      source: ''''

      src: ''''

      relabel: true'
  - name: quadlet_extra_dir_defaults
    type: dict
    description: Values for keys missing in quadlet_extra_dirs entries.
    default: 'mode: ''0755''

      mount: ''''

      source: ''''

      src: ''''

      relabel: true'
